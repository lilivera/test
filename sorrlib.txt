Option Explicit

'==================================================
' Windows API
'==================================================

Private Declare Function AttachConsole Lib "kernel32" _
    (ByVal dwProcessId As Long) As Long

Private Declare Function GetStdHandle Lib "kernel32" _
    (ByVal nStdHandle As Long) As Long

Private Declare Function WriteConsoleW Lib "kernel32" _
    (ByVal hConsoleOutput As Long, _
     ByVal lpBuffer As Long, _
     ByVal nNumberOfCharsToWrite As Long, _
     lpNumberOfCharsWritten As Long, _
     ByVal lpReserved As Long) As Long

Private Declare Sub ExitProcess Lib "kernel32" _
    (ByVal uExitCode As Long)

Private Const ATTACH_PARENT_PROCESS As Long = -1
Private Const STD_OUTPUT_HANDLE As Long = -11


'==================================================
' メイン処理
'==================================================

Public Sub Main()

    Dim filePath As String

    Dim lines() As String
    Dim keys() As String
    Dim cols() As String

    Dim lineText As String

    Dim f As Integer
    Dim count As Long
    Dim capacity As Long
    Dim i As Long

    '----------------------------------------------
    ' 親コンソールへ接続
    '----------------------------------------------
    Call AttachConsole(ATTACH_PARENT_PROCESS)

    On Error GoTo ErrHandler

    '----------------------------------------------
    ' 引数取得
    '----------------------------------------------
    filePath = Trim$(Command$)

    ' "C:\work\data.csv" のように
    ' ダブルクォートで囲まれている場合
    If Len(filePath) >= 2 Then

        If Left$(filePath, 1) = """" And _
           Right$(filePath, 1) = """" Then

            filePath = Mid$(filePath, 2, Len(filePath) - 2)

        End If

    End If

    '----------------------------------------------
    ' 引数チェック
    '----------------------------------------------
    If Len(filePath) = 0 Then

        ConsoleEcho "ERROR: 入力ファイルが指定されていません。"

        ExitProcess 1

    End If

    '----------------------------------------------
    ' ファイル存在チェック
    '----------------------------------------------
    If Dir$(filePath) = "" Then

        ConsoleEcho "ERROR: 入力ファイルが存在しません。"
        ConsoleEcho "FILE : " & filePath

        ExitProcess 2

    End If

    '----------------------------------------------
    ' 初期領域確保
    '----------------------------------------------
    capacity = 1000

    ReDim lines(0 To capacity - 1)
    ReDim keys(0 To capacity - 1)

    count = 0

    '----------------------------------------------
    ' CSV読込
    '----------------------------------------------
    f = FreeFile

    Open filePath For Input As #f

    Do Until EOF(f)

        Line Input #f, lineText

        '------------------------------------------
        ' カンマで分割
        ' 囲い文字なし前提
        '------------------------------------------
        cols = Split(lineText, ",")

        '------------------------------------------
        ' 30項目チェック
        ' 0 ～ 29 なので UBound = 29
        '------------------------------------------
        If UBound(cols) <> 29 Then

            Close #f

            ConsoleEcho _
                "ERROR: CSV項目数不正。" & _
                " 行=" & CStr(count + 1) & _
                " 項目数=" & CStr(UBound(cols) + 1)

            ExitProcess 3

        End If

        '------------------------------------------
        ' 配列拡張
        '------------------------------------------
        If count >= capacity Then

            capacity = capacity + 1000

            ReDim Preserve lines(0 To capacity - 1)
            ReDim Preserve keys(0 To capacity - 1)

        End If

        '------------------------------------------
        ' 元レコードをそのまま保存
        '------------------------------------------
        lines(count) = lineText

        '------------------------------------------
        ' 3項目目をソートキーとして保存
        '
        ' CSV 3項目目
        ' → 配列では cols(2)
        '
        ' 7桁0埋めなので文字列ソートで
        ' 数値順と一致する
        '------------------------------------------
        keys(count) = cols(2)

        count = count + 1

    Loop

    Close #f

    '----------------------------------------------
    ' 0件ファイル
    '----------------------------------------------
    If count = 0 Then

        ConsoleEcho "正常終了: 対象データ 0件"

        ExitProcess 0

    End If

    '----------------------------------------------
    ' 実件数へ配列縮小
    '----------------------------------------------
    ReDim Preserve lines(0 To count - 1)
    ReDim Preserve keys(0 To count - 1)

    '----------------------------------------------
    ' 昇順ソート
    '----------------------------------------------
    If count > 1 Then

        QuickSort _
            lines, _
            keys, _
            0, _
            count - 1

    End If

    '----------------------------------------------
    ' 元ファイルへ上書き
    '----------------------------------------------
    f = FreeFile

    Open filePath For Output As #f

    For i = 0 To count - 1

        Print #f, lines(i)

    Next i

    Close #f

    '----------------------------------------------
    ' 正常終了
    '----------------------------------------------
    ConsoleEcho _
        "正常終了: " & _
        CStr(count) & _
        "件をソートしました。"

    ExitProcess 0

    Exit Sub


'==================================================
' エラー処理
'==================================================
ErrHandler:

    Dim errNo As Long
    Dim errMsg As String

    errNo = Err.Number
    errMsg = Err.Description

    On Error Resume Next

    Close #f

    ConsoleEcho _
        "ERROR: " & _
        CStr(errNo) & _
        " " & _
        errMsg

    ExitProcess 9

End Sub


'==================================================
' QuickSort
'==================================================

Private Sub QuickSort( _
    ByRef lines() As String, _
    ByRef keys() As String, _
    ByVal first As Long, _
    ByVal last As Long)

    Dim i As Long
    Dim j As Long

    Dim pivot As String

    Dim tmpLine As String
    Dim tmpKey As String

    i = first
    j = last

    pivot = keys((first + last) \ 2)

    Do While i <= j

        '------------------------------------------
        ' pivotより小さい間、右へ
        '------------------------------------------
        Do While StrComp( _
            keys(i), _
            pivot, _
            vbBinaryCompare) < 0

            i = i + 1

        Loop

        '------------------------------------------
        ' pivotより大きい間、左へ
        '------------------------------------------
        Do While StrComp( _
            keys(j), _
            pivot, _
            vbBinaryCompare) > 0

            j = j - 1

        Loop

        '------------------------------------------
        ' 入れ替え
        '------------------------------------------
        If i <= j Then

            tmpLine = lines(i)
            lines(i) = lines(j)
            lines(j) = tmpLine

            tmpKey = keys(i)
            keys(i) = keys(j)
            keys(j) = tmpKey

            i = i + 1
            j = j - 1

        End If

    Loop

    '----------------------------------------------
    ' 左側
    '----------------------------------------------
    If first < j Then

        QuickSort _
            lines, _
            keys, _
            first, _
            j

    End If

    '----------------------------------------------
    ' 右側
    '----------------------------------------------
    If i < last Then

        QuickSort _
            lines, _
            keys, _
            i, _
            last

    End If

End Sub


'==================================================
' コンソール出力
'==================================================

Private Sub ConsoleEcho(ByVal message As String)

    Dim hStdOut As Long
    Dim written As Long
    Dim text As String

    text = message & vbCrLf

    hStdOut = GetStdHandle(STD_OUTPUT_HANDLE)

    If hStdOut <> 0 Then

        Call WriteConsoleW( _
            hStdOut, _
            StrPtr(text), _
            Len(text), _
            written, _
            0)

    End If

End Sub
